57void Texture2D::createFromRaw(const void* data, int width, int height,
58 texture_internal_format internalFormat, pixel_format format,
59 pixel_type type, const TextureSettings& settings) const
60{
61 if (width <= 0 || height <= 0)
62 {
63 throw graphics_error(graphics_error_code::texture_load_failed,
64 "Invalid texture dimensions for createFromRaw.");
65 }
66
67 const GLsizei levels =
68 settings.generateMipmaps
69 ? static_cast<GLsizei>(std::floor(std::log2(std::max(width, height))) + 1)
70 : 1;
71
72 // Use underlying values for OpenGL calls
73 glTextureStorage2D(m_Handle.get(), levels, to_underlying(internalFormat), width, height);
74
75 if (data != nullptr)
76 {
77 glTextureSubImage2D(m_Handle.get(), 0, 0, 0, width, height, to_underlying(format),
78 to_underlying(type), data);
79 }
80
81 glTextureParameteri(m_Handle.get(), GL_TEXTURE_WRAP_S, to_underlying(settings.wrapS));
82 glTextureParameteri(m_Handle.get(), GL_TEXTURE_WRAP_T, to_underlying(settings.wrapT));
83 glTextureParameteri(m_Handle.get(), GL_TEXTURE_MIN_FILTER,
84 to_underlying(settings.minFilter));
85 glTextureParameteri(m_Handle.get(), GL_TEXTURE_MAG_FILTER,
86 to_underlying(settings.magFilter));
87
88 if (settings.generateMipmaps && data != nullptr)
89 {
90 glGenerateTextureMipmap(m_Handle.get());
91 }
92}

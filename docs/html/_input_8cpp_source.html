55void update(const gfx::Window& window)
56{
57 GLFWwindow* w = native(window);
58
59 g_State.prevKeys = g_State.keys;
60 g_State.prevButtons = g_State.buttons;
61
62 for (int k = GLFW_KEY_SPACE; k <= GLFW_KEY_LAST; ++k) // keys below SPACE are invalid
63 g_State.keys[k] = glfwGetKey(w, k) == GLFW_PRESS;
64
65 for (int b = 0; b <= GLFW_MOUSE_BUTTON_LAST; ++b)
66 g_State.buttons[b] = glfwGetMouseButton(w, b) == GLFW_PRESS;
67
68 const glm::vec2 now = getCursorPosition(window);
69 // No delta on the first frame, otherwise the camera jumps
70 g_State.delta = g_State.hasCursor ? now - g_State.cursor : glm::vec2{0.f};
71 g_State.cursor = now;
72 g_State.hasCursor = true;
73}

124void updateCamera(gfx::Camera& camera, const gfx::Window& window, float dt, float moveSpeed,
125 float turnSpeed)
126{
127 // Look: arrow keys
128 float yaw = 0.f, pitch = 0.f;
129 if (isKeyDown(window, Key::LEFT))
130 yaw -= 1.f;
131 if (isKeyDown(window, Key::RIGHT))
132 yaw += 1.f;
133 if (isKeyDown(window, Key::UP))
134 pitch += 1.f;
135 if (isKeyDown(window, Key::DOWN))
136 pitch -= 1.f;
137
138 if (yaw != 0.f || pitch != 0.f)
139 {
140 const float newPitch = std::clamp(camera.getPitch() + pitch * turnSpeed * dt, -89.f, 89.f);
141 camera.setRotation(camera.getYaw() + yaw * turnSpeed * dt, newPitch);
142 }
143
144 // Move: WASD + Space/Shift, in camera space
145 glm::vec3 dir{0.f}; // x = right, y = up, z = forward
146 if (isKeyDown(window, Key::W))
147 dir.z += 1.f;
148 if (isKeyDown(window, Key::S))
149 dir.z -= 1.f;
150 if (isKeyDown(window, Key::D))
151 dir.x += 1.f;
152 if (isKeyDown(window, Key::A))
153 dir.x -= 1.f;
154 if (isKeyDown(window, Key::SPACE))
155 dir.y += 1.f;
156 if (isKeyDown(window, Key::LEFT_CTRL))
157 dir.y -= 1.f;
158
159 if (glm::dot(dir, dir) > 0.f)
160 {
161 dir = glm::normalize(dir) * moveSpeed * dt; // same speed on diagonals
162 camera.moveLocal(dir.x, dir.y, dir.z);
163 }
164}

203void update(const gfx::Window& window)
204{
205 GLFWwindow* w = native(window);
206
207 g_State.prevKeys = g_State.keys;
208 g_State.prevButtons = g_State.buttons;
209
210 for (int k = GLFW_KEY_SPACE; k <= GLFW_KEY_LAST; ++k) // keys below SPACE are invalid
211 g_State.keys[k] = glfwGetKey(w, k) == GLFW_PRESS;
212
213 for (int b = 0; b <= GLFW_MOUSE_BUTTON_LAST; ++b)
214 g_State.buttons[b] = glfwGetMouseButton(w, b) == GLFW_PRESS;
215
216 const glm::vec2 now = getCursorPosition(window);
217 // No delta on the first frame, otherwise the camera jumps
218 g_State.delta = g_State.hasCursor ? now - g_State.cursor : glm::vec2{0.f};
219 g_State.cursor = now;
220 g_State.hasCursor = true;
221}

270void updateCamera(gfx::Camera& camera, const gfx::Window& window, float dt, float moveSpeed,
271 float turnSpeed)
272{
273 // Look: arrow keys
274 float yaw = 0.f, pitch = 0.f;
275 if (isKeyDown(window, GLFW_KEY_LEFT))
276 yaw -= 1.f;
277 if (isKeyDown(window, GLFW_KEY_RIGHT))
278 yaw += 1.f;
279 if (isKeyDown(window, GLFW_KEY_UP))
280 pitch += 1.f;
281 if (isKeyDown(window, GLFW_KEY_DOWN))
282 pitch -= 1.f;
283
284 if (yaw != 0.f || pitch != 0.f)
285 {
286 const float newPitch = std::clamp(camera.getPitch() + pitch * turnSpeed * dt, -89.f, 89.f);
287 camera.setRotation(camera.getYaw() + yaw * turnSpeed * dt, newPitch);
288 }
289
290 // Move: WASD + Space/Shift, in camera space
291 glm::vec3 dir{0.f}; // x = right, y = up, z = forward
292 if (isKeyDown(window, GLFW_KEY_W))
293 dir.z += 1.f;
294 if (isKeyDown(window, GLFW_KEY_S))
295 dir.z -= 1.f;
296 if (isKeyDown(window, GLFW_KEY_D))
297 dir.x += 1.f;
298 if (isKeyDown(window, GLFW_KEY_A))
299 dir.x -= 1.f;
300 if (isKeyDown(window, GLFW_KEY_SPACE))
301 dir.y += 1.f;
302 if (isKeyDown(window, GLFW_KEY_LEFT_CONTROL))
303 dir.y -= 1.f;
304
305 if (glm::dot(dir, dir) > 0.f)
306 {
307 dir = glm::normalize(dir) * moveSpeed * dt; // same speed on diagonals
308 camera.moveLocal(dir.x, dir.y, dir.z);
309 }
310}

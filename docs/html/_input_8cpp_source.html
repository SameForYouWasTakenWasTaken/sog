51void update(const gfx::Window& window)
52{
53 GLFWwindow* w = native(window);
54
55 g_State.prevKeys = g_State.keys;
56 g_State.prevButtons = g_State.buttons;
57
58 for (int k = GLFW_KEY_SPACE; k <= GLFW_KEY_LAST; ++k) // keys below SPACE are invalid
59 g_State.keys[k] = glfwGetKey(w, k) == GLFW_PRESS;
60
61 for (int b = 0; b <= GLFW_MOUSE_BUTTON_LAST; ++b)
62 g_State.buttons[b] = glfwGetMouseButton(w, b) == GLFW_PRESS;
63
64 const glm::vec2 now = getCursorPosition(window);
65 // No delta on the first frame, otherwise the camera jumps
66 g_State.delta = g_State.hasCursor ? now - g_State.cursor : glm::vec2{0.f};
67 g_State.cursor = now;
68 g_State.hasCursor = true;
69}

118void updateCamera(gfx::Camera& camera, const gfx::Window& window, float dt, float moveSpeed,
119 float turnSpeed)
120{
121 // Look: arrow keys
122 float yaw = 0.f, pitch = 0.f;
123 if (isKeyDown(window, GLFW_KEY_LEFT))
124 yaw -= 1.f;
125 if (isKeyDown(window, GLFW_KEY_RIGHT))
126 yaw += 1.f;
127 if (isKeyDown(window, GLFW_KEY_UP))
128 pitch += 1.f;
129 if (isKeyDown(window, GLFW_KEY_DOWN))
130 pitch -= 1.f;
131
132 if (yaw != 0.f || pitch != 0.f)
133 {
134 const float newPitch = std::clamp(camera.getPitch() + pitch * turnSpeed * dt, -89.f, 89.f);
135 camera.setRotation(camera.getYaw() + yaw * turnSpeed * dt, newPitch);
136 }
137
138 // Move: WASD + Space/Shift, in camera space
139 glm::vec3 dir{0.f}; // x = right, y = up, z = forward
140 if (isKeyDown(window, GLFW_KEY_W))
141 dir.z += 1.f;
142 if (isKeyDown(window, GLFW_KEY_S))
143 dir.z -= 1.f;
144 if (isKeyDown(window, GLFW_KEY_D))
145 dir.x += 1.f;
146 if (isKeyDown(window, GLFW_KEY_A))
147 dir.x -= 1.f;
148 if (isKeyDown(window, GLFW_KEY_SPACE))
149 dir.y += 1.f;
150 if (isKeyDown(window, GLFW_KEY_LEFT_CONTROL))
151 dir.y -= 1.f;
152
153 if (glm::dot(dir, dir) > 0.f)
154 {
155 dir = glm::normalize(dir) * moveSpeed * dt; // same speed on diagonals
156 camera.moveLocal(dir.x, dir.y, dir.z);
157 }
158}

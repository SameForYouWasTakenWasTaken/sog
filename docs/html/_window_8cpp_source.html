139void Window::initialize(const context::WindowContext& context)
140{
141 if (!glfwInit())
142 throw graphics_error(graphics_error_code::window_creation_failed,
143 "Failed to create window! (glfw did not initialize properly)");
144 glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, context.OpenGL_Version_Major);
145 glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, context.OpenGL_Version_Minor);
146 glfwWindowHint(GLFW_OPENGL_PROFILE, backend::toGLFWProfile(context.OpenGL_Profile));
147 glfwWindowHint(GLFW_OPENGL_DEBUG_CONTEXT, GLFW_TRUE);
148
149 m_Window->glfwHandle =
150 glfwCreateWindow(context.width, context.height, context.title.c_str(), nullptr, nullptr);
151 if (!m_Window->glfwHandle)
152 {
153 glfwTerminate();
154 throw graphics_error(graphics_error_code::window_creation_failed,
155 "Failed to create window!");
156 }
157 glfwMakeContextCurrent(m_Window->glfwHandle);
158 glfwSetWindowUserPointer(m_Window->glfwHandle, this);
159 glfwSetFramebufferSizeCallback(m_Window->glfwHandle, ImplWindow::on_resize);
160
161 if (!gladLoadGL(glfwGetProcAddress))
162 {
163 // Constructor is about to throw: clean up the already-created window so it is not leaked
164 // (~Window never runs for a failed construction).
165 glfwDestroyWindow(m_Window->glfwHandle);
166 m_Window->glfwHandle = nullptr;
167 glfwTerminate();
168 throw graphics_error(graphics_error_code::window_creation_failed,
169 "Failed to create window! (glad did not initialize properly)");
170 }
171}

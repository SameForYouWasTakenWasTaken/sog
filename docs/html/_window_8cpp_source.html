136void Window::initialize(const context::WindowContext& context)
137{
138 if (!glfwInit())
139 throw graphics_error(graphics_error_code::window_creation_failed,
140 "Failed to create window! (glfw did not initialize properly)");
141 glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, context.OpenGL_Version_Major);
142 glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, context.OpenGL_Version_Minor);
143 glfwWindowHint(GLFW_OPENGL_PROFILE, backend::toGLFWProfile(context.OpenGL_Profile));
144 glfwWindowHint(GLFW_OPENGL_DEBUG_CONTEXT, GLFW_TRUE);
145
146 m_Window->glfwHandle =
147 glfwCreateWindow(context.width, context.height, context.title.c_str(), nullptr, nullptr);
148 if (!m_Window->glfwHandle)
149 {
150 glfwTerminate();
151 throw graphics_error(graphics_error_code::window_creation_failed,
152 "Failed to create window!");
153 }
154 glfwMakeContextCurrent(m_Window->glfwHandle);
155 glfwSetWindowUserPointer(m_Window->glfwHandle, this);
156 glfwSetFramebufferSizeCallback(m_Window->glfwHandle, ImplWindow::on_resize);
157
158 if (!gladLoadGL(glfwGetProcAddress))
159 {
160 throw graphics_error(graphics_error_code::window_creation_failed,
161 "Failed to create window! (glad did not initialize properly)");
162 }
163}

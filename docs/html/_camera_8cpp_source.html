15// Rotation that maps +Y onto 'up'
22glm::quat frameFromUp(const math::Vec3& up)
23{
24 const math::Vec3 y{0.f, 1.f, 0.f};
25 const float d = glm::dot(y, up);
26 if (d > 0.9999f)
27 return glm::quat{1.f, 0.f, 0.f, 0.f};
28 if (d < -0.9999f)
29 return glm::angleAxis(glm::pi<float>(), math::Vec3{1.f, 0.f, 0.f});
30 return glm::angleAxis(std::acos(d), glm::normalize(glm::cross(y, up)));
31}

39Camera::Basis Camera::basis() const
40{
41 const float yaw = glm::radians(m_Yaw), pitch = glm::radians(m_Pitch);
42
43 const math::Vec3 f = m_Frame * math::Vec3{std::cos(pitch) * std::cos(yaw), std::sin(pitch),
44 std::cos(pitch) * std::sin(yaw)};
45 // Depends only on yaw, so it stays valid at any pitch
46 math::Vec3 r = m_Frame * math::Vec3{-std::sin(yaw), 0.f, std::cos(yaw)};
47 math::Vec3 u = glm::cross(r, f);
48
49 if (m_Roll != 0.f)
50 {
51 const glm::quat roll = glm::angleAxis(glm::radians(m_Roll), f);
52 r = roll * r;
53 u = roll * u;
54 }
55 return {f, r, u};
56}

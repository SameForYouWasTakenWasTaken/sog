136 frameBuffer.attachTexture(gfx::FramebufferAttachment::Color0, colorTexture.getHandle());
137 frameBuffer.attachRenderbuffer(gfx::FramebufferAttachment::DepthStencil, depthStencilBuffer);
138
139 gfx::VertexShader frameBufVertShader;
140 frameBufVertShader.createFromFile("assets/shaders/PPO.vert");
141 gfx::FragmentShader frameBufFragShader;
142 frameBufFragShader.createFromFile("assets/shaders/PPO.frag");
143
144 gfx::ShaderProgram frameBufShaderProgram;
145 frameBufShaderProgram.Initialize(frameBufVertShader, frameBufFragShader);
146
147 auto locFramebufTexture = frameBufShaderProgram.getLocation("screenTexture");
148 auto locFramebufTime = frameBufShaderProgram.getLocation("time");
149 auto locFramebufIntensity = frameBufShaderProgram.getLocation("intensity");
150 gfx::uniform::setInt(frameBufShaderProgram, locFramebufTexture.value_or(gfx::Uniform{}), 0);
151
152 assert(frameBuffer.isComplete());
153 while (!window.shouldClose())
154 {
155 auto currentFrame = static_cast<float>(Time::getTime());
156 deltaTime = currentFrame - lastFrame;
157 lastFrame = currentFrame;
158
159 gfx::Event e{};
160 while (window.pollEvent(e))
161 {
162 if (e.type == gfx::EventType::Resized)
163 {
164 int width = e.size.width;
165 int height = e.size.height;
166
167 // Resize your FBO resources seamlessly right here!
168 depthStencilBuffer = gfx::RenderBuffer(GL_DEPTH24_STENCIL8, width, height);
169 colorTexture = gfx::Texture2D{};
170 colorTexture.createFromRaw(nullptr, width, height, GL_RGBA8, GL_RGBA,
171 GL_UNSIGNED_BYTE, {});
172
173 frameBuffer.attachTexture(gfx::FramebufferAttachment::Color0,
174 colorTexture.getHandle());
175 frameBuffer.attachRenderbuffer(gfx::FramebufferAttachment::DepthStencil,
176 depthStencilBuffer);
177 }
178 }
179
180 input::update(window);
181 input::updateCamera(camera, window, deltaTime);
182
183 frameBuffer.bind();
184 glViewport(0, 0, window.getWidth(), window.getHeight());
185 gfx::Window::clear(gfx::ClearBuffer::Color | gfx::ClearBuffer::Depth,
186 gfx::Color{255.f, 255.f, 255.f, 255.f});
187
188 // drawing
189 gfx::uniform::setMat4(shaderProgram, locMVP, camera.getViewProjection() * transform);
190 gfx::Window::draw(gfx::RenderState{vertexBuffer, vertexArray, &texture, shaderProgram});
191
192 // framebuffer
193 gfx::uniform::runSafe(
194 [&]
195 {
196 gfx::uniform::setFloat(frameBufShaderProgram,
197 locFramebufTime.value_or(gfx::Uniform{}), currentFrame);
198 },
199 [&]
200 {
201 gfx::uniform::setFloat(frameBufShaderProgram,
202 locFramebufIntensity.value_or(gfx::Uniform{}), 0.2f);
203 });
204 // apply post processing
205 gfx::FrameBuffer::unbind();
206 window.beginFrame(gfx::ClearBuffer::Color | gfx::ClearBuffer::Depth,
207 gfx::Color{255, 0, 0, 0});
208 frameBufShaderProgram.use();
209 colorTexture.bind(0);
210 glDrawArrays(GL_TRIANGLES, 0, 3);
211 window.endFrame();
212 }
213 return 0;
214}

113[[nodiscard]] ObjectBuffers createObjectBuffers(const Object& object)
114{
115 ObjectBuffers buffers;
116
117 buffers.vertexBuffer.set(object.vertices);
118 buffers.indexBuffer.set(object.indices);
119
120 buffers.vertexArray.create();
121 buffers.vertexArray.setVertexBuffer(buffers.vertexBuffer, 0, sog::Vertex::stride());
122 buffers.vertexArray.setIndexBuffer(buffers.indexBuffer);
123 buffers.vertexArray.setAttributes(sog::Vertex::attributes());
124
125 return buffers;
126}

128void prepareObject(const Object& object, sog::ShaderProgram& program, const sog::Texture2D& texture,
129 const sog::math::Mat4& viewProjection)
130{
131 const auto mvpLocation = sog::uniform::getLocation(program, "MVP");
132 const auto textureLocation = sog::uniform::getLocation(program, "texture2D");
133
134 sog::uniform::setMat4(program, mvpLocation, viewProjection * object.getModelMatrix());
135 sog::uniform::setInt(program, textureLocation, 0);
136
137 texture.bind(0);
138}

167int Application::Run()
168{
169 sog::Window window;
170
171 Object object;
172 auto buffers = createObjectBuffers(object);
173
174 sog::VertexShader vertexShader{std::filesystem::path("assets/shaders/vertex.glsl")};
175 sog::FragmentShader fragmentShader{std::filesystem::path("assets/shaders/fragment.glsl")};
176 sog::ShaderProgram shaderProgram{vertexShader, fragmentShader};
177
178 sog::Texture2D texture;
179 texture.createFromFile("assets/Textures/test.jpg", {.wrapS = sog::gfx::texture_params::REPEAT,
180 .wrapT = sog::gfx::texture_params::REPEAT});
181
182 object.width = 1.f;
183 object.height = 1.f;
184 object.depth = 1.f;
185
186 auto transform = glm::mat4{1.f};
187 transform = glm::rotate(transform, glm::radians(90.f), glm::vec3{0.f, 0.f, 1.f});
188
189 sog::Camera camera;
190
191 float lastFrame = 0.f;
192
193 sog::gfx::SetRenderingState(sog::gfx::RenderSettings{.depthTest = true, .blending = true});
194
195 sog::RenderBuffer depthStencilBuffer(GL_DEPTH24_STENCIL8, 800, 600);
196 auto colorTexture = sog::Texture2D::empty(800, 600);
197
198 sog::FrameBuffer frameBuffer;
199 frameBuffer.attachTexture(sog::gfx::FramebufferAttachment::Color0, colorTexture.getHandle());
200 frameBuffer.attachRenderbuffer(sog::gfx::FramebufferAttachment::DepthStencil,
201 depthStencilBuffer);
202
203 sog::VertexShader frameBufVertShader{std::filesystem::path("assets/shaders/PPO.vert")};
204 sog::FragmentShader frameBufFragShader{std::filesystem::path("assets/shaders/PPO.frag")};
205
206 sog::ShaderProgram frameBufShaderProgram;
207 frameBufShaderProgram.Initialize(frameBufVertShader, frameBufFragShader);
208
209 const auto locFramebufTexture = frameBufShaderProgram.getLocation("screenTexture");
210 const auto locFramebufTime = frameBufShaderProgram.getLocation("time");
211 const auto locFramebufIntensity = frameBufShaderProgram.getLocation("intensity");
212
213 assert(frameBuffer.isComplete());
214
215 while (!window.shouldClose())
216 {
217 const auto currentFrame = static_cast<float>(sog::Time::getTime());
218 const auto deltaTime = currentFrame - lastFrame;
219 lastFrame = currentFrame;
220
221 sog::Event event{};
222 while (window.pollEvent(event))
223 {
224 if (event.type == sog::EventType::Resized)
225 {
226 const int width = event.size.width;
227 const int height = event.size.height;
228
229 if (width > 0 && height > 0)
230 {
231 resizeFramebuffer(frameBuffer, depthStencilBuffer, colorTexture, width, height);
232 }
233 }
234 }
235
236 sog::input::update(window);
237 sog::input::updateCamera(camera, window, deltaTime);
238
239 // Scene pass: render the cube into the framebuffer.
240 frameBuffer.bind();
241
242 sog::ViewportToActiveTarget(0, 0, window.getWidth(), window.getHeight());
243
244 sog::Window::clear(sog::ClearBuffer::Color | sog::ClearBuffer::Depth,
245 sog::Color{255.f, 255.f, 255.f, 255.f});
246
247 prepareObject(object, shaderProgram, texture, camera.getViewProjection() * transform);
248
249 sog::Window::draw(sog::gfx::RenderState{buffers.vertexBuffer, buffers.vertexArray, &texture,
250 shaderProgram});
251
252 // Post-processing pass.
253 updatePostProcessing(frameBufShaderProgram, locFramebufTime, locFramebufIntensity,
254 currentFrame);
255
256 sog::FrameBuffer::unbind();
257
258 window.beginFrame(sog::ClearBuffer::Color | sog::ClearBuffer::Depth,
259 sog::Color{255, 0, 0, 0});
260
261 frameBufShaderProgram.use();
262 colorTexture.bind(0);
263
264 sog::DrawArrays(sog::enums::draw_mode::TRIANGLES, 0, 3);
265
266 window.endFrame();
267 }
268
269 return 0;
270}
